cmake_minimum_required(VERSION 3.20)
project(CyxWizLibArchiveConfigTest NONE)
# Fixture only: exercise Windows layout without a compiler or installed SDK.
set(WIN32 TRUE)
set(prefix "${CMAKE_CURRENT_BINARY_DIR}/fixture")
file(MAKE_DIRECTORY "${prefix}/include" "${prefix}/lib" "${prefix}/debug/lib"
    "${prefix}/share/libarchive")
file(WRITE "${prefix}/include/archive.h" "#define ARCHIVE_VERSION_ONLY_STRING \"3.8.4\"\n")
file(WRITE "${prefix}/lib/archive.lib" "release fixture")
file(WRITE "${prefix}/debug/lib/archive.lib" "debug fixture")
file(WRITE "${prefix}/share/libarchive/vcpkg-cmake-wrapper.cmake" "# fixture")
set(LibArchive_INCLUDE_DIR "${prefix}/include" CACHE PATH "" FORCE)
set(LibArchive_LIBRARY "${prefix}/debug/lib/archive.lib" CACHE FILEPATH "" FORCE)
include("${CMAKE_CURRENT_LIST_DIR}/../../CyxWizLibArchive.cmake")
foreach(config IN ITEMS DEBUG RELEASE)
    get_target_property(location LibArchive::LibArchive IMPORTED_LOCATION_${config})
    if(config STREQUAL "DEBUG")
        set(expected "${prefix}/debug/lib/archive.lib")
    else()
        set(expected "${prefix}/lib/archive.lib")
    endif()
    if(NOT location STREQUAL expected)
        message(FATAL_ERROR "${config}: ${location} != ${expected}")
    endif()
endforeach()
foreach(config IN ITEMS RELWITHDEBINFO MINSIZEREL)
    get_target_property(mapping LibArchive::LibArchive MAP_IMPORTED_CONFIG_${config})
    if(NOT mapping STREQUAL "RELEASE")
        message(FATAL_ERROR "${config} must map to Release")
    endif()
endforeach()
if(NOT LibArchive_LIBRARY STREQUAL "${prefix}/lib/archive.lib")
    message(FATAL_ERROR "Stale Debug cache was not repaired")
endif()
# Re-inclusion must retain the Debug mapping even after repairing the cache.
include("${CMAKE_CURRENT_LIST_DIR}/../../CyxWizLibArchive.cmake")
get_target_property(debug LibArchive::LibArchive IMPORTED_LOCATION_DEBUG)
if(NOT debug STREQUAL "${prefix}/debug/lib/archive.lib")
    message(FATAL_ERROR "Repeated discovery lost Debug mapping")
endif()
message(STATUS "LibArchive configuration contract passed")
