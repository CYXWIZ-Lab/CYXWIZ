cmake_minimum_required(VERSION 3.20)
project(CyxWizDataConvertTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()

# Small service-only harness: no Engine, ArrayFire, Python or shared build outputs.
get_filename_component(CYXWIZ_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Arrow CONFIG REQUIRED)
find_package(Parquet CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(HighFive CONFIG QUIET)
include("${CYXWIZ_ROOT}/cmake/CyxWizDataConvert.cmake")
if(NOT TARGET Arrow::arrow_shared AND TARGET Arrow::arrow_static)
    add_library(Arrow::arrow_shared ALIAS Arrow::arrow_static)
endif()
if(NOT TARGET Parquet::parquet_shared AND TARGET Parquet::parquet_static)
    add_library(Parquet::parquet_shared ALIAS Parquet::parquet_static)
endif()

add_library(data_convert_under_test STATIC
    ../../src/core/data_convert_service.cpp
    ../../src/core/arrow_dataset.cpp)
target_include_directories(data_convert_under_test PUBLIC ../../src "${CYXWIZ_ROOT}/cyxwiz-backend/include")
target_link_libraries(data_convert_under_test PUBLIC
    Arrow::arrow_shared Parquet::parquet_shared spdlog::spdlog nlohmann_json::nlohmann_json)
cyxwiz_target_data_convert(data_convert_under_test)
if(HighFive_FOUND)
    target_link_libraries(data_convert_under_test PUBLIC HighFive)
    target_compile_definitions(data_convert_under_test PUBLIC CYXWIZ_HAS_HDF5)
endif()
if(MSVC)
    target_compile_definitions(data_convert_under_test PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(data_convert_under_test PRIVATE /W4)
endif()

foreach(test_name IN ITEMS test_data_convert_service test_excel_table_adapter test_data_convert_reload test_data_convert_formats)
    add_executable(${test_name} ../${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE data_convert_under_test)
    if(TARGET OpenXLSX::OpenXLSX)
        target_link_libraries(${test_name} PRIVATE OpenXLSX::OpenXLSX)
        target_compile_definitions(${test_name} PRIVATE CYXWIZ_HAS_XLSX)
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
target_sources(test_excel_table_adapter PRIVATE ../test_xlsx_archive_guard.cpp)
if(TARGET OpenXLSX::OpenXLSX)
    target_link_libraries(test_excel_table_adapter PRIVATE LibArchive::LibArchive)
endif()

# Headless rendering only; no window, graphics device or Engine is launched.
find_package(imgui CONFIG REQUIRED)
add_executable(test_data_convert_format_controls ../test_data_convert_format_controls.cpp)
target_link_libraries(test_data_convert_format_controls PRIVATE imgui::imgui)
add_test(NAME test_data_convert_format_controls COMMAND test_data_convert_format_controls)
